# Dependency directions. Names are packages relative to internal/, groups
# defined here, or (with a dot) the import-path prefix of an external module.
#
# group NAME: members
# forbid FROM: targets   FROM may not import any target
# only PACKAGE: imports  PACKAGE may import nothing else from this module
# violation FROM: target an existing breach of a forbid rule; these only shrink

group transport: app/server graphql
group bootstrap: app app/bootstrap
group core: database
group checker-execution: jobs/checker jobs/checker/judges jobs/queue/proxy
group listeners: rotatingproxy
group persistence: database platform/pg platform/schema gorm.io github.com/jackc/pgx
# Base packages extracted from the core. They sit below it.
group base: platform/pg platform/schema evidence
# Feature packages. Add each new feature package here in the change that creates it.
group features: abuseipdb account alerts blacklist devseed geolite ipgeolocation judgestore mail scrapestore settings workspace

# Base packages never import the core, a feature, transport or bootstrap.
forbid base: core features transport bootstrap auth
# Feature packages never import transport, authentication middleware or bootstrap.
forbid features: transport bootstrap auth
# alerts and workspace sit below the core: the core may call them, they never
# import the core.
forbid alerts: core
forbid workspace: core
# geolite, blacklist, scrapestore and devseed sit above the core: they may
# call the core, the core never imports them.
forbid core: geolite blacklist scrapestore devseed
# The core never imports transport, bootstrap, checker execution or listeners.
forbid core: transport bootstrap auth checker-execution listeners
# Compiled checker settings, judge selection and cadence never import
# persistence; loaders are injected by app/bootstrap.
forbid checkerconfig: persistence
forbid jobs/checker/judges: persistence
forbid checkpace: persistence
# checkpace receives persistence through Configure. The core and the hot path
# may import it.
only checkpace: domain

violation database: jobs/checker/judges  # checker settings read the judge cache (checker_settings.go); checker-settings SQL stays in the core
